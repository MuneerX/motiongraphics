import { parseIntent } from '../lib/whisper/parseIntent';
import type { VoiceQuery } from '../types';

export const useVoiceQuery = () => {
  const queryRef = useRef<VoiceQuery>({});
  const [searchParams, setSearchParams]
    = useState<VoiceQuery>({});
  const deferred
    = useDeferredValue(searchParams);
  const [isPending, startTransition]
    = useTransition();

  const onTranscript = (t: string) => {
    const next = parseIntent(t);
    queryRef.current = { ...queryRef.current, ...next };
    startTransition(() => {
      setSearchParams(queryRef.current);
    });
  };
  return { deferred, isPending, onTranscript };
};
